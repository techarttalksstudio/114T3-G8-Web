float balloonY = 600;
float balloonX = 400;
int score = 0;
void setup(){
size(800,600);
}
void draw(){
background(#F0F8FF);
fill(#ACD6FF);
ellipse(balloonX,balloonY,100,100);//ellipse(x,y,width,height);
balloonY = balloonY - 3;
if(balloonY < -50){
balloonY = 650;
}
if(balloonY < -50){
balloonY = 650;
balloonX = random(50,750);
}
text("Score: " + score,20,40);
fill(0);
textSize(30);
text("Score: " + score, 20, 40);
if(dist(mouseX, mouseY, balloonX, balloonY)<50){
score++;
balloonY = 650;
balloonX = random(50,750);
}
}
