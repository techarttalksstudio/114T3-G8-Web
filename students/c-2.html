import processing.sound.*;
float balloonY = 600;
float balloonX = 400;
int score = 0;
SoundFile popSound;
void setup(){
size(800,600);
popSound = new SoundFile(this, "pop.mp3");
}
void draw(){
background(255);
fill(#AD5A5A);
ellipse(balloonX,balloonY,100,100);//x,y,width,height
balloonY = balloonY - 3;
if(balloonY < -50){
balloonY = 650;
balloonX = random(50,750);
}
fill(0);
textSize(30);
text("score="+score, 20,40);
}
void mousePressed(){
if (dist(mouseX, mouseY, balloonX, balloonY)<50){
score++; //score = score +1
balloonY = 650;
balloonX = random(50,750);
popSound.play();
}
}
